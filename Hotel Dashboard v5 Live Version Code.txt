        // ... existing parseCSV function ...
        function parseCSV(csvText) {
            const rows = [];
            let currentRow = [];
            let currentField = '';
            let inQuotes = false;
            // Handle BOM (Byte Order Mark)
            if (csvText.charCodeAt(0) === 0xFEFF) {
                csvText = csvText.substring(1);
            }
            csvText += '\n'; // Add newline to ensure last row is processed
            for (let i = 0; i < csvText.length; i++) {
                const char = csvText[i];
                const nextChar = csvText[i + 1];
                if (inQuotes) {
                    if (char === '"') {
                        if (nextChar === '"') {
                            currentField += '"';
                            i++; // Skip next quote
                        } else {
                            inQuotes = false;
                        }
                    } else {
                        currentField += char;
                    }
                } else {
                    if (char === '"') {
                        inQuotes = true;
                    } else if (char === ',') {
                        currentRow.push(currentField);
                        currentField = '';
                    } else if (char === '\n' || char === '\r') {
                        if (char === '\r' && nextChar === '\n') {
                            i++; // Skip CRLF
                        }
                        currentRow.push(currentField);
                        // Only add row if it has some data
                        const hasData = currentRow.some(field => field !== '');
                        if (hasData) {
                            rows.push(currentRow);
                        }
                        currentRow = [];
                        currentField = '';
                    } else {
                        currentField += char;
                    }
                }
            }
            return rows;
        }

        // --- UPDATED: processData function ---
        function processData(csvText) {
            // Clear old dynamic cards before (re)processing
            // This ensures if data is re-loaded, it doesn't stack
            const dynamicCards = document.querySelectorAll('.dynamic-card');
            dynamicCards.forEach(card => card.remove());

            // Clear charts
            if (arrivalChartInstance) arrivalChartInstance.destroy();
            if (departureChartInstance) departureChartInstance.destroy();

            const rows = parseCSV(csvText);
            if (rows.length <= 1) {
                throw new Error("CSV file is empty or has no data rows.");
            }
            const totalDataRows = rows.length - 1;
            const headers = rows[0].map(h => h.trim().replace(/^"|"$/g, ''));
            
            const indices = {
                country: headers.indexOf(COLUMNS.COUNTRY),
                status: headers.indexOf(COLUMNS.DELEGATION_STATUS),
                hotel: headers.indexOf(COLUMNS.HOTEL),
                arrivalDate: headers.indexOf(COLUMNS.ARRIVAL_DATE),
                arrivalTime: headers.indexOf(COLUMNS.ARRIVAL_TIME), 
                departureDate: headers.indexOf(COLUMNS.DEPARTURE_DATE),
                departureTime: headers.indexOf(COLUMNS.DEPARTURE_TIME)
            };
            
            const requiredCols = [
                { key: 'country', name: COLUMNS.COUNTRY },
                { key: 'status', name: COLUMNS.DELEGATION_STATUS },
                { key: 'hotel', name: COLUMNS.HOTEL },
                { key: 'arrivalDate', name: COLUMNS.ARRIVAL_DATE },
                { key: 'arrivalTime', name: COLUMNS.ARRIVAL_TIME },
                { key: 'departureDate', name: COLUMNS.DEPARTURE_DATE },
                { key: 'departureTime', name: COLUMNS.DEPARTURE_TIME }
            ];

            const missing = requiredCols
                .filter(col => indices[col.key] === -1)
                .map(col => `"${col.name}"`)
                .join(', ');

            if (missing) {
                throw new Error(`CSV missing required columns: ${missing}`);
            }

            const participants = [];
            for (let i = 1; i < rows.length; i++) {
                let hotel_raw = (rows[i][indices.hotel] || '').trim().replace(/^"|"$/g, ''); 
                let country = (rows[i][indices.country] || '').trim().replace(/^"|"$/g, '');
                let status = (rows[i][indices.status] || '').trim().replace(/^"|"$/g, '');
                let arrivalDate = (rows[i][indices.arrivalDate] || '').trim().replace(/^"|"$/g, '');
                let arrivalTime = (rows[i][indices.arrivalTime] || '').trim().replace(/^"|"$/g, '');
                let departureDate = (rows[i][indices.departureDate] || '').trim().replace(/^"|"$/g, '');
                let departureTime = (rows[i][indices.departureTime] || '').trim().replace(/^"|"$/g, '');

                if (!country) country = 'Unknown';
                if (!status) status = 'Unknown';

                const hotelBin = getDesignatedHotelBin(hotel_raw);
                
                participants.push({ 
                    hotel: hotelBin, 
                    country, status, 
                    arrivalDate, arrivalTime, 
                    departureDate, departureTime 
                });
            }

            const hotelCounts = countBy(participants, 'hotel');
            const countryCounts = countBy(participants, 'country');

            const { arrivalDateCounts } = processArrivalData(participants);
            const { departureDateCounts } = processDepartureData(participants);
            
            const logisticsGridData = processLogisticsGrid(participants);
            const departureLogisticsGridData = processDepartureLogisticsGrid(participants);

            renderKPIs(
                totalDataRows, hotelCounts, countryCounts, 
                arrivalDateCounts,
                departureDateCounts,
                logisticsGridData, 
                departureLogisticsGridData 
            );
        }

        // --- NEW: Helper function to get time bin ---
        function getTimeBin(timeStr) {
            if (!timeStr) return "Not Specified";

            try {
                // This is surprisingly robust for "10:00 AM", "10:30", "14:00"
                const date = new Date('1970-01-01 ' + timeStr); 
                if (!isNaN(date.getTime())) {
                    const hour = date.getHours();
                    if (hour >= 6 && hour < 12) return "Morning (6am-12pm)";
                    if (hour >= 12 && hour < 18) return "Afternoon (12pm-6pm)";
                    // UPDATED: Combined Evening (18-24) and Night (0-6)
                    if ((hour >= 18 && hour < 24) || (hour >= 0 && hour < 6)) return "Evening (6pm-6am)";
                }

                // Fallback for "1000H"
                const match = timeStr.match(/^(\d{2})(\d{2})H?$/);
                if (match) {
                     const hour = parseInt(match[1]);
                     if (hour >= 6 && hour < 12) return "Morning (6am-12pm)";
                     if (hour >= 12 && hour < 18) return "Afternoon (12pm-6pm)";
                     // UPDATED: Combined Evening (18-24) and Night (0-6)
                     if ((hour >= 18 && hour < 24) || (hour >= 0 && hour < 6)) return "Evening (6pm-6am)";
                }
            } catch (e) {
                // fall through
            }
            return "Not Specified";
        }

        // --- NEW: Helper function to parse date string ---
        function parseDateString(dateStr) {
            try {
                // --- Manual parsing for "month/day/year" format ---
                const parts = dateStr.split('/');
                let date;
                
                // Check for "month/day/year" with 2 or 4 digit year
                if (parts.length === 3) {
                    const month = parseInt(parts[0], 10);
                    const day = parseInt(parts[1], 10);
                    let year = parseInt(parts[2], 10);

                    // Handle two-digit year, assume 20xx
                    if (year < 100) {
                        year += 2000;
                    }
                    
                    // new Date(year, monthIndex, day)
                    // month is 0-indexed (0=Jan, 11=Dec)
                    date = new Date(year, month - 1, day);
                } else {
                   // Try default parsing as a fallback if format is not as expected
                   date = new Date(dateStr);
                }

                if (date && !isNaN(date.getTime())) {
                    return date;
                }
                return null; // Invalid date
            } catch (e) {
                return null; // Error parsing
            }
        }

        // --- NEW: Single source of truth for hotel categorization ---
        function getDesignatedHotelBin(hotelName) {
            const lowerHotel = (hotelName || '').trim().toLowerCase();

            // 1. Check for non-hotel terms FIRST
            if (NON_HOTEL_TERMS.has(lowerHotel)) {
                return 'Other / No Hotel';
            }

            // 2. Check for designated hotels
            if (lowerHotel.includes('dusit thani')) return 'Dusit Thani';
            if (lowerHotel.includes('go hotels')) return 'Go Hotels';
            if (lowerHotel.includes('park inn')) return 'Park Inn';
            if (lowerHotel.includes('acacia hotel')) return 'Acacia Hotel';
            if (lowerHotel.includes('discovery samal')) return 'Waterfront Hotel'; // Map Discovery
            if (lowerHotel.includes('waterfront hotel')) return 'Waterfront Hotel';

            // 3. Everything else (e.g., other hotels not on the list) is 'Other / No Hotel'
            return 'Other / No Hotel';
        }

        // --- NEW: Helper function to process Master Logistics Grid data ---
        function processLogisticsGrid(participants) {
            const gridData = {}; // { 'date': { 'timeBin': { 'hotelBin': count } } }
            const grandTotals = { byHotel: {}, byTime: {}, total: 0 };
            
            // Initialize grand totals
            MATRIX_HOTELS.forEach(h => grandTotals.byHotel[h] = 0);
            MATRIX_TIME_BINS.forEach(t => grandTotals.byTime[t] = 0);

            participants.forEach(p => {
                const date = parseDateString(p.arrivalDate);
                // Skip participants with no valid arrival date
                if (!date) return; 

                const dateKey = date.toLocaleDateString('en-US', { year: 'numeric', month: 'long', day: 'numeric' });
                const timeBin = getTimeBin(p.arrivalTime); // Use existing robust function
                
                // Skip participants with a timebin not in our matrix list (if any)
                if (!MATRIX_TIME_BINS.includes(timeBin)) return;

                const hotelBin = p.hotel;

                // If the hotel is 'Other / No Hotel', skip this participant for the grid
                if (hotelBin === 'Other / No Hotel') return; 

                // Initialize objects if they don't exist
                if (!gridData[dateKey]) {
                    gridData[dateKey] = { _dateObj: date, _dayTotals: { byHotel: {}, total: 0 } };
                    MATRIX_HOTELS.forEach(h => gridData[dateKey]._dayTotals.byHotel[h] = 0);
                }
                if (!gridData[dateKey][timeBin]) {
                    gridData[dateKey][timeBin] = { _rowTotal: 0 };
                    MATRIX_HOTELS.forEach(h => gridData[dateKey][timeBin][h] = 0);
                }

                // Increment counts
                gridData[dateKey][timeBin][hotelBin]++;
                gridData[dateKey][timeBin]._rowTotal++;
                gridData[dateKey]._dayTotals.byHotel[hotelBin]++;
                gridData[dateKey]._dayTotals.total++;

                // Increment grand totals
                grandTotals.byHotel[hotelBin]++;
                grandTotals.byTime[timeBin]++;
                grandTotals.total++;
            });

            // Sort dates
            const sortedDateKeys = Object.keys(gridData).sort((a, b) => gridData[a]._dateObj - gridData[b]._dateObj);
            
            return { sortedData: gridData, sortedDateKeys, grandTotals };
        }

        // --- NEW: Count-up animation function ---
        function animateCountUp(elementId, endValue) {
            const element = document.getElementById(elementId);
            if (!element) return;

            const startValue = 0;
            const duration = 1000; // 1 second animation
            let startTime = null;

            function animationStep(timestamp) {
                if (!startTime) startTime = timestamp;
                const progress = timestamp - startTime;
                const percentage = Math.min(progress / duration, 1);
                
                // Calculate the current value
                const currentValue = Math.floor(percentage * (endValue - startValue) + startValue);
                
                element.textContent = currentValue;

                if (percentage < 1) {
                    // Continue animation
                    requestAnimationFrame(animationStep);
                } else {
                    // Ensure it ends on the exact value
                    element.textContent = endValue; 
                }
            }
            // Start the animation
            requestAnimationFrame(animationStep);
        }

        function renderMasterLogisticsGrid(data) {
            const { sortedData, sortedDateKeys, grandTotals } = data;
            const container = document.getElementById('logistics-matrix-container');

            if (sortedDateKeys.length === 0) {
                container.innerHTML = '<p class="text-gray-500 text-center">No valid arrival data found for the grid.</p>';
                return;
            }

            // UPDATED: Filter out the "Not Specified" bin from rendering
            const binsToRender = MATRIX_TIME_BINS.filter(bin => bin !== "Not Specified");
            const rowSpan = binsToRender.length;

            let tableHtml = '<table class="bi-matrix-table">';
            
            // 1. Table Header
            tableHtml += '<thead><tr>';
            // UPDATED: Column Titles
            tableHtml += '<th class="sticky-col header-cell">Date of Arrival</th>';
            tableHtml += '<th class="sticky-col header-cell">Time Period</th>';
            
            // --- START OF USER REQUESTED CHANGE ---
            // This loop now only iterates over the 5 hotels in the updated MATRIX_HOTELS array
            MATRIX_HOTELS.forEach(hotel => {
                tableHtml += `<th class="header-cell">${hotel}</th>`;
            });
            // --- END OF USER REQUESTED CHANGE ---

            tableHtml += '<th class="total-col header-cell">TOTAL</th>';
            tableHtml += '</tr></thead>';

            // 2. Table Body
            tableHtml += '<tbody>';
            sortedDateKeys.forEach((dateKey) => {
                const dateData = sortedData[dateKey];
                
                // UPDATED: Create date label with weekday
                const dateObj = new Date(dateData._dateObj);
                const shortDate = dateObj.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
                const weekDay = dateObj.toLocaleDateString('en-US', { weekday: 'long' });
                const dateLabel = `${shortDate}<br>(${weekDay})`; // e.g., "Nov 9<br>(Sunday)"

                // UPDATED: Loop over filtered bins
                binsToRender.forEach((timeBin, timeIndex) => {
                    const timeData = dateData[timeBin];
                    const isFirstRowForDate = timeIndex === 0;

                    // Add thick border class for the first row of a new date
                    tableHtml += `<tr class="${isFirstRowForDate ? 'date-row-start' : ''}">`;

                    // Rowspan for Date Cell (only on the first time bin)
                    if (isFirstRowForDate) {
                        // UPDATED: Use new dateLabel and corrected rowSpan
                        tableHtml += `<td class="sticky-col date-cell" rowspan="${rowSpan}">${dateLabel}</td>`;
                    }
                    
                    // Time Bin Cell
                    tableHtml += `<td class="sticky-col time-cell">${timeBin.split(' ')[0]}</td>`; // Just "Morning", "Afternoon" etc.

                    // --- START OF USER REQUESTED CHANGE ---
                    // Hotel Data Cells - This loop now only iterates over the 5 designated hotels
                    MATRIX_HOTELS.forEach(hotel => {
                        const count = timeData ? (timeData[hotel] || 0) : 0;
                        tableHtml += `<td class="data-cell">${count === 0 ? 0 : count}</td>`;
                    });
                    // --- END OF USER REQUESTED CHANGE ---

                    // Row Total Cell
                    const rowTotal = timeData ? (timeData._rowTotal || 0) : 0;
                    tableHtml += `<td class="total-col data-cell">${rowTotal === 0 ? 0 : rowTotal}</td>`;

                    tableHtml += '</tr>';
                });

                // Add Day Total Row
                const dayTotals = dateData._dayTotals;
                tableHtml += '<tr class="day-total-row">';
                tableHtml += '<td class="sticky-col total-cell" colspan="2">Day Total</td>'; // Colspan 2 (Date + Time)
                
                // --- START OF USER REQUESTED CHANGE ---
                // Day Totals - This loop now only iterates over the 5 designated hotels
                MATRIX_HOTELS.forEach(hotel => {
                    const count = dayTotals.byHotel[hotel] || 0;
                    tableHtml += `<td class="total-cell data-cell">${count === 0 ? 0 : count}</td>`;
                });
                // --- END OF USER REQUESTED CHANGE ---
                
                tableHtml += `<td class="total-col total-cell data-cell">${dayTotals.total === 0 ? 0 : dayTotals.total}</td>`;
                tableHtml += '</tr>';
            });

            // 3. Table Footer (Grand Totals)
            tableHtml += '<tfoot class="grand-total-row">';
            tableHtml += '<tr>';
            tableHtml += '<td class="sticky-col total-cell" colspan="2">GRAND TOTAL</td>';
            
            // --- START OF USER REQUESTED CHANGE ---
            // Grand Totals - This loop now only iterates over the 5 designated hotels
            MATRIX_HOTELS.forEach(hotel => {
                const count = grandTotals.byHotel[hotel] || 0;
                tableHtml += `<td class="total-cell data-cell">${count === 0 ? 0 : count}</td>`;
            });
            // --- END OF USER REQUESTED CHANGE ---

            tableHtml += `<td class="total-col total-cell data-cell">${grandTotals.total === 0 ? 0 : grandTotals.total}</td>`;
            tableHtml += '</tr>';
            tableHtml += '</tfoot>';

            tableHtml += '</tbody></table>';
            container.innerHTML = tableHtml;
        }

        // --- NEW: Function to load and process data from Google Sheets ---
        async function loadDashboardData() {
            // Get elements
            const loadingSpinner = document.getElementById('loading-spinner');
            const errorEl = document.getElementById('error-message');
            const loadingState = document.getElementById('loading-state');

            // Check if the user has updated the URL
            if (GOOGLE_SHEET_CSV_URL === 'PASTE_YOUR_GOOGLE_SHEET_CSV_URL_HERE' || GOOGLE_SHEET_CSV_URL.trim() === '') {
                errorEl.textContent = 'Configuration Error: Please update the GOOGLE_SHEET_CSV_URL variable in the HTML script.';
                loadingSpinner.classList.add('hidden');
                return;
            }

            try {
                // 1. Fetch data from the live URL
                // Note: Added a cache-busting parameter to help bypass Google's 5-min cache.
                const response = await fetch(GOOGLE_SHEET_CSV_URL.replace('/pub?', '/pub?output=csv&') + '&t=' + new Date().getTime());

                if (!response.ok) {
                    throw new Error(`Failed to fetch data. Status: ${response.status}. Check the URL, network, or sheet permissions.`);
                }

                const csvText = await response.text();
                
                if (csvText.trim() === "") {
                    throw new Error("The fetched CSV data is empty. Check the Google Sheet.");
                }

                // 2. Process the data (using the existing function)
                processData(csvText);

                // 3. Show the dashboard
                dashboard.classList.remove('hidden');

                // 4. Animate the new content
                const newItems = document.querySelectorAll('#dashboard-container h2, #dashboard-container .bg-white.rounded-lg.shadow');
                newItems.forEach((item, index) => {
                    item.classList.add('animated-item');
                    item.style.animationDelay = `${index * 75}ms`;
                });
                
                // 5. Hide the loading state
                loadingState.classList.add('hidden');

            } catch (err) {
                // Handle errors
                console.error('Error loading dashboard data:', err);
                errorEl.textContent = `Error: ${err.message}. Could not load live data.`;
                if (loadingSpinner) loadingSpinner.classList.add('hidden');
            }
        }
